## 2. 数学工具（Geometry.ts）

### 2.1 点到线段距离

```typescript
function pointToSegmentDist(p: Vec2, s: LineSegment): number {
    const dx = s.p2.x - s.p1.x;
    const dy = s.p2.y - s.p1.y;
    const len2 = dx * dx + dy * dy;
    if (len2 === 0) return Math.hypot(p.x - s.p1.x, p.y - s.p1.y);
    let t = ((p.x - s.p1.x) * dx + (p.y - s.p1.y) * dy) / len2;
    t = Math.max(0, Math.min(1, t));
    const px = s.p1.x + t * dx;
    const py = s.p1.y + t * dy;
    return Math.hypot(p.x - px, p.y - py);
}
```

### 2.2 线段-线段相交（拼刀核心）

```typescript
function segmentsIntersect(a: LineSegment, b: LineSegment): boolean {
    const d1 = cross(b.p1, b.p2, a.p1);
    const d2 = cross(b.p1, b.p2, a.p2);
    const d3 = cross(a.p1, a.p2, b.p1);
    const d4 = cross(a.p1, a.p2, b.p2);
    return ((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) &&
           ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0));
}
function cross(o: Vec2, a: Vec2, b: Vec2): number {
    return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
}
```

### 2.3 扇形-圆相交（扫掠命中）

```typescript
function sectorCircleIntersect(sec: Sector, cir: Circle): boolean {
    const d = Math.hypot(cir.c.x - sec.center.x, cir.c.y - sec.center.y);
    // 半径范围检查
    if (d > sec.r1 + cir.r || d < Math.max(sec.r0 - cir.r, 0)) return false;
    // 角度范围检查
    let ang = Math.atan2(cir.c.y - sec.center.y, cir.c.x - sec.center.x);
    ang = normalizeAngle(ang);
    return angleInRange(ang, sec.angleStart, sec.angleEnd);
}
```

## 3. 空间分区 Grid（SpatialGrid.ts）

class SpatialGrid {
    private cellSize: number;
    private cols: number;
    private rows: number;
    private cells: Map<number, Entity[]>;

    constructor(width: number, height: number, cellSize = 120) {
        this.cellSize = cellSize;
        this.cols = Math.ceil(width / cellSize);
        this.rows = Math.ceil(height / cellSize);
        this.cells = new Map();
    }

    clear(): void { this.cells.clear(); }

    insert(entity: Entity, bounds: AABB): void {
        const minC = Math.floor(bounds.x / this.cellSize);
        const minR = Math.floor(bounds.y / this.cellSize);
        const maxC = Math.floor((bounds.x + bounds.w) / this.cellSize);
        const maxR = Math.floor((bounds.y + bounds.h) / this.cellSize);
        for (let r = minR; r <= maxR; r++)
            for (let c = minC; c <= maxC; c++)
                this.getOrCreate(r * this.cols + c).push(entity);
    }

    query(bounds: AABB): Entity[] {
        const result = new Set<Entity>();
        // 同 insert 的范围遍历，收集去重
        // ...
        return [...result];
    }
}

## 4. 碰撞引擎主流程（CollisionEngine.ts）

```typescript
class CollisionEngine implements ISystem {
    private grid: SpatialGrid;
    private blades: BladeEntity[] = [];      // 玩家+敌方刀体
    private enemies: Enemy[] = [];
    private listener: ICollisionListener;

    update(dt: number, ctx: GameContext): void {
        // 1. 更新刀体线段端点（基于旋转角）
        for (const b of this.blades) b.updateEndpoints();
        // 2. 重建网格
        this.grid.clear();
        for (const e of this.enemies) this.grid.insert(e, e.bounds);
        for (const b of this.blades) this.grid.insert(b, b.bounds);
        // 3. 刀-圆检测（命中）
        this.checkBladeHits(dt);
        // 4. 刀-刀检测（拼刀）
        this.checkBladeClashes();
        // 5. 实体-地形检测（阻挡）
        this.checkBodyBlocks();
    }

    private checkBladeHits(dt: number): void {
        for (const blade of this.blades) {
            const nearby = this.grid.query(blade.sweepBounds);
            for (const e of nearby) {
                // 扇形扫掠检测（防穿透）
                if (sectorCircleIntersect(blade.sweepSector, e.circle)) {
                    if (blade.cooldown <= 0) {
                        this.listener.onBladeHitEnemy(blade, e, blade.tipPos);
                        blade.cooldown = 0.1;   // 命中CD防同帧多次
                    }
                }
            }
        }
    }

    private checkBladeClashes(): void {
        // 仅检测玩家刀 vs 敌方刀
        const playerBlades = this.blades.filter(b => b.owner === 'player');
        const foeBlades = this.blades.filter(b => b.owner === 'enemy');
        for (const pb of playerBlades) {
            for (const fb of foeBlades) {
                // 距离粗筛
                if (dist(pb.center, fb.center) > pb.length + fb.length) continue;
                if (pb.clashCooldown > 0 || fb.clashCooldown > 0) continue;
                if (segmentsIntersect(pb.segment, fb.segment)) {
                    this.listener.onBladeClash(pb, fb, intersectPoint(pb.segment, fb.segment));
                    pb.clashCooldown = fb.clashCooldown = 1.2;
                }
            }
        }
    }
}
```

## 5. 拼刀解算（ClashResolver.ts）

```typescript
class ClashResolver {
    resolve(pb: BladeEntity, fb: BladeEntity, ctx: GameContext): void {
        const Mp = bladeMomentum(pb);    // L*W*ω*Q
        const Mf = bladeMomentum(fb);
        let winRate = clamp(Mp / (Mp + Mf), 0.1, 0.9);
        // 修正项
        if (pb.counterRotating(fb)) winRate += 0.05;        // 相向旋转
        if (fb.justDrawn(0.25)) winRate += 0.05;            // 抓出刀窗口
        if (fb.comboBuff) winRate -= 0.05;
        winRate = clamp(winRate, 0.1, 0.9);

        const roll = ctx.rng.next();
        if (roll < winRate) {
            // 玩家胜
            if (Mp >= Mf * 2 && pb.owner.hasTalent('破刀诀') && ctx.rng.next() < 0.3) {
                this.triggerBladeBreak(pb, fb);             // 破刀
            } else {
                this.triggerStagger(fb, 1.5);               // 敌僵直
                fb.disableBlade(2.0);
                pb.owner.applyDebuff('破势', 1.5);          // 后续命中×1.5
            }
        } else if (roll > 1 - (1 - winRate) * 0.5) {
            this.triggerStagger(pb.owner, 0.8);             // 玩家败
            pb.disableBlade(1.0);
        } else {
            this.triggerStagger(pb.owner, 0.4);             // 平
            this.triggerStagger(fb, 0.4);
        }
        // 拼刀经验
        ctx.player.addExp(25, '刀法');
    }
}
```
